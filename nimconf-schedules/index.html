Every NimConf, in tables

NimConf is the Nim community's own conference: online, free, talks recorded beforehand and premiered on YouTube while their speakers answer questions in the chat. There have been five, and each published its schedule as a table. Here they all are again, made with tabelline, the display tables for Nim that just reached 0.1: from the simplest table to the two tracks of 2026, then where the data comes from, and the Nim code that downloaded it.

That is all it takes: tabella turns any sequence of objects (or named tuples) into a table, one column per field, and nbTable puts it in the page. It is honest and plain. A schedule wants a bit more: a title, only the columns a reader needs, the talk linked to its video, the length of each video (read from each edition's YouTube playlist, see step 6 of How the data was downloaded: length maps a video's id to its seconds), and the breaks set apart. Each of those is one verb, the same verbs as R's gt and Python's Great Tables, in camelCase:

type Talk = object
  kind, start, talk, speakers: string
  length: Option[int]          # seconds; none for a break

func link(s: Slot): string =
  ## the title, linked to its video, in Markdown
  let title = s.title.replace("*", "\\*")       # Nim*AI is not emphasis
  if s.video.len == 0: title else: &"[{title}](https://youtu.be/{s.video})"

proc talks(year: int, track = ""): seq[Talk] =
  for s in slots:
    if s.year != year or s.track != track: continue
    result.add Talk(kind: s.kind, start: s.start, talk: s.link, speakers: s.speakers,
                    length: (if s.video in length: some(length[s.video]) else: none(int)))

Two tracks: NimConf 2026

In 2026 there were so many talks that NimConf ran two tracks at once, Projects and Social and Workflow, each its own YouTube playlist. Its page is a different table: one row per starting time, one column per track, the speaker written before the title. The data keeps one slot per talk, with its track; turning the tracks into columns is a few lines of Nim, and tabelline puts a spanner over them:

type Row = object
  start, projects, social: string

let year = editions[^1]
var rows: seq[Row]
for s in slots.filterIt(it.year == 2026):
  if rows.len == 0 or rows[^1].start != s.start: rows.add Row(start: s.start)

Five editions side by side

With every slot in one list, a table can count across the editions: how many talks and speakers each had, how long its videos run in all, and its longest talk.

type Summary = object
  edition: int
  day: string
  talks, speakers: int
  video: int
  longest: string

var summary: seq[Summary]
for e in editions:
  let mine = slots.filterIt(it.year == e.year and it.kind == "talk")
  var people: HashSet[string]
  for s in mine:
    for p in s.speakers.people: people.incl p
  let timed = mine.filterIt(it.video in length)
  let long = timed.sortedByIt(-length[it.video])[0]
  summary.add Summary(edition: e.year, day: e.day.rsplit(' ', 1)[0], talks: mine.len,
                      speakers: people.len, video: timed.mapIt(length[it.video]).foldl(a + b),

2. Download a page, once

download gets a page with std/httpclient and keeps a copy in cache/, so the page is read from the network only the first time (and this page can run the code below while it is built). The proxy line is there because this was written in a sandbox whose traffic goes through one.

nimconf.nim

proc download*(url: string): string =
  ## The page at `url`, downloaded once: later calls read the copy in `cache/`.
  let file = cacheDir / url.multiReplace(("https://", ""), ("/", "_"), ("?", "_"), ("=", "_"))
  if fileExists(file): return readFile(file)
  let proxy = getEnv("HTTPS_PROXY")   # the sandbox this was written in goes through a proxy
  let client = newHttpClient(proxy = (if proxy.len > 0: newProxy(proxy) else: nil))
  defer: client.close()
  result = client.getContent(url)
  createDir(cacheDir)
  writeFile(file, result)

3. Read a piece of HTML

The pages are small and regular, so there is no HTML parser: three procs on strings are enough. between yields every piece of text between two markers (every <tr> of a table, every <td> of a row), plain strips the tags from a piece and decodes the usual entities, and dropComments removes the <!-- ... --> of a page.

nimconf.nim

iterator between*(s, open, close: string): string =
  ## Every piece of `s` that starts after `open` and ends before the next `close`.
  var i = s.find(open)
  while i >= 0:
    let start = s.find('>', i) + 1          # `open` can be `<td`: skip its attributes
    let stop = s.find(close, start)
    if stop < 0: break
    yield s[start ..< stop]
    i = s.find(open, stop)

func plain*(html: string): string =
  ## The text of a piece of HTML: no tags, the usual entities decoded, spaces collapsed.
  var inTag = false
  for c in html:
    if c == '<': inTag = true
    elif c == '>': inTag = false
    elif not inTag: result.add c
  result = result.multiReplace(("&amp;", "&"), ("&lt;", "<"), ("&gt;", ">"), ("&quot;", "\""),
                               ("&#39;", "'"), ("&nbsp;", " ")).splitWhitespace.join(" ")

func dropComments*(html: string): string =
  ## The page without its `<!-- ... -->`: the 2026 page keeps the 2024 schedule in one.
  var i = 0
  while i < html.len:
    let open = html.find("<!--", i)
    if open < 0:
      result.add html[i .. ^1]; break
    result.add html[i ..< open]
    let close = html.find("-->", open)

const cacheDir* = currentSourcePath().parentDir / "cache"

proc download*(url: string): string =
  ## The page at `url`, downloaded once: later calls read the copy in `cache/`.
  let file = cacheDir / url.multiReplace(("https://", ""), ("/", "_"), ("?", "_"), ("=", "_"))
  if fileExists(file): return readFile(file)
  let proxy = getEnv("HTTPS_PROXY")   # the sandbox this was written in goes through a proxy
  let client = newHttpClient(proxy = (if proxy.len > 0: newProxy(proxy) else: nil))
  defer: client.close()
  result = client.getContent(url)
  createDir(cacheDir)
  writeFile(file, result)

# -------------------------------------------------------------------------------- reading HTML

iterator between*(s, open, close: string): string =
  ## Every piece of `s` that starts after `open` and ends before the next `close`.
  var i = s.find(open)
  while i >= 0:
    let start = s.find('>', i) + 1          # `open` can be `<td`: skip its attributes
    let stop = s.find(close, start)
    if stop < 0: break
    yield s[start ..< stop]
    i = s.find(open, stop)

func dropComments*(html: string): string =
  ## The page without its `<!-- ... -->`: the 2026 page keeps the 2024 schedule in one.
  var i = 0
  while i < html.len:
    let open = html.find("<!--", i)
    if open < 0:
      result.add html[i .. ^1]; break
    result.add html[i ..< open]
    let close = html.find("-->", open)
    i = if close < 0: html.len else: close + 3

func plain*(html: string): string =
  ## The text of a piece of HTML: no tags, the usual entities decoded, spaces collapsed.
  var inTag = false
  for c in html:
    if c == '<': inTag = true
    elif c == '>': inTag = false
    elif not inTag: result.add c
  result = result.multiReplace(("&amp;", "&"), ("&lt;", "<"), ("&gt;", ">"), ("&quot;", "\""),
                               ("&#39;", "'"), ("&nbsp;", " ")).splitWhitespace.join(" ")

func videoId*(html: string): string =
  ## The YouTube id in the first link of `html` (`watch?v=<id>&...`), or "".
  let i = html.find("watch?v=")
  if i >= 0: html[i + 8 ..< i + 19] else: ""

type
  Slot* = object
    year*: int
    track*: string      ## "" when the edition had one track
    start*: string      ## UTC, "11:00"
    kind*: string       ## talk, break or end
    title*: string
    speakers*: string
    video*: string      ## YouTube id, "" when the page links none

func slotKind*(title: string): string =
  let t = title.toLowerAscii
  if t.contains("break"): "break" elif t == "track end": "end" else: "talk"

func scheduleTable*(page: string): string =
  ## The schedule's `<table>`: the first one after its heading, comments dropped.
  let html = page.dropComments
  let at = html.find("id=\"time-schedule\"")
  for table in html[max(at, 0) .. ^1].between("<table", "</table>"):
    return table

func oneTrack*(year: int, table: string): seq[Slot] =
  ## 2020 to 2024: a row per talk, cells Talk name (with its video link), Starting time,
  ## Presenter. A break is a row with one cell.
  var start = ""
  for row in table.between("<tr", "</tr>"):
    let cells = toSeq(row.between("<td", "</td>"))
    if cells.len == 0: continue                       # the header row has <th>
    let title = cells[0].plain.replace(" (video)", "").replace(" (link)", "")
    if cells.len >= 3: start = cells[1].plain.replace(" UTC", "")
    result.add Slot(year: year, start: (if cells.len >= 3: start else: ""), kind: slotKind(title),
                    title: title, speakers: (if cells.len >= 3: cells[2].plain else: ""),
                    video: cells[0].videoId)

func splitSpeaker*(cell: string): (string, string) =
  ## 2026 writes "Speaker: Title" in one cell; a cell with no colon has no speaker.
  let i = cell.find(':')
  if i < 0 or cell.slotKind != "talk": ("", cell) else: (cell[0 ..< i].strip, cell[i + 1 .. ^1].strip)

func tracks*(year: int, table: string): seq[Slot] =
  ## 2026: a row per starting time, a column per track (the header names them).
  let names = toSeq(table.between("<th>", "</th>")).mapIt(it.plain.replace(" track", ""))
  for row in table.between("<tr", "</tr>"):
    let cells = toSeq(row.between("<td", "</td>"))
    if cells.len == 0: continue
    for col in 1 ..< cells.len:
      let text = cells[col].plain
      if text.len == 0: continue
      let (speaker, title) = splitSpeaker(text)
      result.add Slot(year: year, track: names[col], start: cells[0].plain, kind: slotKind(title),
                      title: title, speakers: speaker, video: cells[col].videoId)

func schedule*(year: int, page: string): seq[Slot] =
  ## The schedule of an edition from its page, whichever of the two layouts it has.
  let table = scheduleTable(page)
  if table.contains("Talk name"): oneTrack(year, table) else: tracks(year, table)

type

proc jsonText(s: string, at: int): string =
  ## The JSON string that starts at `at` (just after its opening quote), unescaped.
  var i = at
  while i < s.len and s[i] != '"':
    i += (if s[i] == '\\': 2 else: 1)
  parseJson("\"" & s[at ..< i] & "\"").getStr

func seconds*(length: string): int =
  ## "1:02:03" or "45:25" as seconds.
  for part in length.split(':'): result = result * 60 + part.parseInt

proc playlistVideos*(page, playlist: string): seq[Video] =
  ## The videos of a playlist page: YouTube puts its data in the page as JSON, one
  ## `lockupViewModel` per video, with its id, its title and the length on its thumbnail.
  const id = "\"contentId\":\""
  const title = "\"title\":{\"content\":\""
  const badge = "\"thumbnailBadgeViewModel\":{\"text\":\""
  for part in page.split("\"lockupViewModel\":{")[1 .. ^1]:
    let (i, t, b) = (part.find(id), part.find(title), part.find(badge))
    if i < 0 or t < 0 or b < 0: continue
    result.add Video(id: part.jsonText(i + id.len), title: part.jsonText(t + title.len),
                     seconds: part.jsonText(b + badge.len).seconds, playlist: playlist)

func normal(s: string): string =
  for c in s.toLowerAscii:
    if c.isAlphaNumeric: result.add c

func videoTalk*(title: string): string =
  ## The talk in a video's title: "Speaker: Title (NimConf 2026)" is "Title".
  result = title
  let paren = result.rfind(" (NimConf")
  if paren >= 0: result = result[0 ..< paren]
  let colon = result.find(": ")
  if colon >= 0: result = result[colon + 2 .. ^1]

proc procSource*(file, name: string): string =
  ## The source of the top-level proc, func, iterator or const `name` in `file`, with its doc comment:
  ## from its line to the next line that starts at column 0.
  let lines = readFile(file).splitLines
  var start = -1
  for i, line in lines:
    for kind in ["proc ", "func ", "iterator ", "const "]:
      if line.startsWith(kind & name & "*") or line.startsWith(kind & name & "("):
        start = i
    if start >= 0: break
  if start < 0: raise newException(KeyError, name & " is not in " & file)
  var stop = start + 1
  while stop < lines.len and (lines[stop].len == 0 or lines[stop][0] == ' '): inc stop
  while stop > start and lines[stop - 1].strip.len == 0: dec stop
  lines[start ..< stop].join("\n")

func scoped(rules, scope: string): string =
  ## Every rule of `rules` under `scope` (rules are one per line or block, as written above).
  for line in rules.splitLines:
    if line.strip.len == 0: continue
